function loadMessages() { const messagesRef = ref(db, 'messages'); document.getElementById('messages').innerHTML = ''; onChildAdded(messagesRef, (snapshot) => { const msg = snapshot.val(); if (!msg.name && !msg.user) return; let timeStr = ''; if (msg.time) { const date = new Date(msg.time); timeStr = date.toLocaleTimeString('uz-UZ', { hour: '2-digit', minute: '2-digit' }); } const div = document.createElement('div'); div.innerHTML = `${msg.name || msg.user}${msg.text}${timeStr}`; document.getElementById('messages').appendChild(div); }); }
